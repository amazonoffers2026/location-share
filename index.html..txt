  <script>
    function shareLocation() {
      const result = document.getElementById("result");

      if (!navigator.geolocation) {
        result.textContent = "Location isn't supported by this browser.";
        return;
      }

      result.textContent = "Please choose Allow if you want to share your location.";

      navigator.geolocation.getCurrentPosition(
        function(position) {
          const lat = position.coords.latitude;
          const lon = position.coords.longitude;

          result.innerHTML =
            '<a href="https://www.google.com/maps?q=' +
            lat + ',' + lon +
            '" target="_blank">Open your location in Google Maps</a>';
        },
        function() {
          result.textContent = "Location was not shared.";
        }
      );
    }
  </script>
