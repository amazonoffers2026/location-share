<!DOCTYPE html>
<html>
<head>
  <title>My Location Page</title>
</head>

<body>
  <h1>Share Your Location</h1>

  <p>
    If you choose to share your location, your browser will ask for permission.
  </p>

  <button onclick="shareLocation()">Share My Location</button>

  <p id="result"></p>

  <script>
    function shareLocation() {
      const result = document.getElementById("result");

      if (!navigator.geolocation) {
        result.textContent = "Location is not supported by this browser.";
        return;
      }

      result.textContent = "Waiting for permission...";

      navigator.geolocation.getCurrentPosition(
        function(position) {
          const latitude = position.coords.latitude;
          const longitude = position.coords.longitude;

          const map =
            "https://www.google.com/maps?q=" +
            latitude + "," + longitude;

          result.innerHTML =
            '<a href="' + map + '" target="_blank">' +
            'Open your location in Google Maps' +
            '</a>';
        },

        function() {
          result.textContent =
            "Location was not shared.";
        }
      );
    }
  </script>
</body>
</html>
